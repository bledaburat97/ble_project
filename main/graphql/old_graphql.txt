    @SuppressWarnings("unchecked")
    private List<EventSummaryProjection> readResultListToEventSummaryProjectionList(String query, Map<String, Object> paramMap, int page, int size, DateFilterEnum dateFilterEnum) {
        // we sure return is tuple list, suppress warnings.
        query += dateFilterEnum.equals(DateFilterEnum.PREVIOUS) ? NativeQuerySql.EVENT.ORDER_BY_START_DATE_DESC : NativeQuerySql.EVENT.ORDER_BY_START_DATE_ASC;
        Query nativeQuery = entityManager.createNativeQuery(query, Tuple.class).setFirstResult(page * size).setMaxResults(size);
        paramMap.forEach(nativeQuery::setParameter);
        List<Tuple> tuples = (List<Tuple>) nativeQuery.getResultList();
        return tuples.stream().map(tuple -> new EventSummaryProjection(
                (Long) tuple.get("event_id"),
                (String) tuple.get("name"),
                (String) tuple.get("description"),
                (Instant) tuple.get("start_date"),
                (Instant) tuple.get("end_date"),
                Enum.valueOf(EventStatus.class, (String) tuple.get("status")),
                (UUID) tuple.get("image_id"),
                (UUID) tuple.get("owner_user_id"),
                (UUID) tuple.get("venue_id"),
                (UUID) tuple.get("address_id"),
                (Integer) tuple.get("max_image_pixel")
        )).toList();
    }

    @Override
    public List<EventSummaryProjection> getEventsByRole(RoleEnum role, EventStatus desiredStatus,
                                                        UUID userId, String keyword, CustomPageable pageable, DateFilterEnum dateFilterEnum, boolean getOnlyVenueIsNotOwnerEvents) {
        Map<String, Object> paramMap = new HashMap<>() {{
            put("userId", userId);
        }};
        String whereCondition = "";
        if(role == RoleEnum.ORGANIZER) {
            whereCondition = NativeQuerySql.EVENT.ORGANIZER_USER_CONDITION;
        }
        else if(role == RoleEnum.VENUE) {
            whereCondition = NativeQuerySql.EVENT.VENUE_USER_CONDITION;
        }
        else if(role == RoleEnum.ARTIST) {
            whereCondition = NativeQuerySql.EVENT.ARTIST_USER_CONDITION;
        }

        String query = NativeQuerySql.EVENT.BASE_GET_EVENT_QUERY.replace("{CONDITION}", whereCondition);

        query = insertDateFilter(dateFilterEnum, query, paramMap);

        if (desiredStatus != null) {
            query += NativeQuerySql.EVENT.EVENT_STATUS_CONDITION;
            paramMap.put("status", desiredStatus.name());
        }

        if (getOnlyVenueIsNotOwnerEvents) {
            query += NativeQuerySql.EVENT.VENUE_IS_NOT_EVENT_OWNER;
        }

        if (keyword != null && !keyword.isEmpty()) {
            query += NativeQuerySql.EVENT.USER_EVENTS_SEARCH_CONDITION;
            paramMap.put("keyword", "%" + keyword + "%");
        }

        return readResultListToEventSummaryProjectionList(query, paramMap, pageable.page(), pageable.size(), dateFilterEnum);
    }

    private List<EventSummaryProjection> findEventSummaryProjections(EventStatus desiredStatus, String keyword, int page, int size, UserVerificationDto userVerification, boolean getOnlyVenueIsNotOwnerEvents, DateFilterEnum dateFilterEnum) {
        List<EventSummaryProjection> events = null;
        if (userVerification.isOrganizer()) {
            if(dateFilterEnum == DateFilterEnum.PREVIOUS){
                logger.error("organizer previous");
            }
            else if(dateFilterEnum == DateFilterEnum.UPCOMING){
                logger.error("organizer upcoming");
            }
            events = eventRepository.getEventsByRole(RoleEnum.ORGANIZER, desiredStatus, userVerification.getOwnerUserId(), keyword, new CustomPageable(page, size), dateFilterEnum, false);
        } else if (userVerification.isVenue()) {
            if(dateFilterEnum == DateFilterEnum.PREVIOUS){
                logger.error("venue previous");
            }
            else if(dateFilterEnum == DateFilterEnum.UPCOMING){
                logger.error("venue upcoming");
            }
            events = eventRepository.getEventsByRole(RoleEnum.VENUE, desiredStatus, userVerification.getOwnerUserId(), keyword, new CustomPageable(page, size), dateFilterEnum, getOnlyVenueIsNotOwnerEvents);
        } else if (userVerification.isArtist()) {
            if(dateFilterEnum == DateFilterEnum.PREVIOUS){
                logger.error("artist previous");
            }
            else if(dateFilterEnum == DateFilterEnum.UPCOMING){
                logger.error("artist upcoming");
            }
            events = eventRepository.getEventsByRole(RoleEnum.ARTIST, desiredStatus, userVerification.getOwnerUserId(), keyword, new CustomPageable(page, size), dateFilterEnum, false);
        }
        return events;
    }
    
    @Override
    @Loggable(description = "to get signed in user events")
    public List<EventSummaryOutDashboard> getSignedInUserEvents(int page, int size, String extUserId, String keyword, DateFilterEnum dateFilterEnum) {
        UserVerificationDto userVerification = userServiceInteractor.verifyBusinessUserById(extUserId);
        if (userVerification == null) {
            logger.error("userVerification is null for [extUserId:%s]".formatted(extUserId));
            return List.of();
        }
        List<EventSummaryProjection> events = eventRepository.getEventsByRole(RoleEnum.ORGANIZER, null, userVerification.getOwnerUserId(), keyword, new CustomPageable(page, size), dateFilterEnum, false);
        return events == null ? List.of() : events.stream().map(eventMapper::eventSummaryProjectionToProEventSummaryOut).toList();
    }

    @Override
    @Loggable(description = "to get user included events")
    public List<EventSummaryOut> getUserIncludedEvents(int page, int size, String extUserId, String keyword, DateFilterEnum dateFilterEnum) {
        UserVerificationDto userVerification = userServiceInteractor.verifyBusinessUserById(extUserId);
        if (userVerification == null) {
            return List.of();
        }
        List<EventSummaryProjection> events = findEventSummaryProjections(EventStatus.APPROVED, keyword, page, size, userVerification, true, dateFilterEnum);
        return events == null ? List.of() : events.stream().map(eventMapper::eventSummaryProjectionToEventSummaryOut).toList();
    }
